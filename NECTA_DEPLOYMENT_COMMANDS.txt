================================================================================
TESTING WORKFLOWS
================================================================================

# Test SCHOOL candidate
1. Candidates > Add Candidate
2. Select: SCHOOL candidate type
3. Choose: A combination (e.g., Science)
4. Save
5. Verify: Subjects auto-allocated from combination

# Test PRIVATE candidate
1. Candidates > Add Candidate
2. Select: PRIVATE candidate type
3. Leave: Combination blank
4. Save
5. Click: Allocate Subjects button
6. Manually: Select subjects (GS 111 required, 3+ principals)
7. Verify: Allocation applied

================================================================================
QUICK DIAGNOSTIC COMMANDS
================================================================================

# Check candidate types exist
mysql -u root -p irms -e "SELECT DISTINCT candidate_type FROM candidates LIMIT 10;"

# Check allocations with sources
mysql -u root -p irms -e "SELECT source, COUNT(*) FROM candidate_subject_selections WHERE source IS NOT NULL GROUP BY source;"

# Check candidates missing General Studies
mysql -u root -p irms -e "SELECT c.id FROM candidates c WHERE c.id NOT IN (SELECT DISTINCT css.candidate_id FROM candidate_subject_selections css JOIN subjects s ON s.id = css.subject_id WHERE s.code = '111');"

# Check for duplicate allocations
mysql -u root -p irms -e "SELECT candidate_id, subject_id, COUNT(*) FROM candidate_subject_selections GROUP BY candidate_id, subject_id HAVING COUNT(*) > 1;"

php artisan db:show
