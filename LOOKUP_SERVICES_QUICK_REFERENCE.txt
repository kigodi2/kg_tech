THREE MAIN LOOKUP SERVICES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. AverageMarksService
   For calculating and looking up average marks and grades
   
2. GradeLookupService
   For looking up division, GPA, competence, and other grade info
   
3. NectaGradingService (Existing)
   The underlying service with NECTA grading rules

File: app/Services/Results/AverageMarksService.php

Main Methods:
  ✓ calculateAverage(float, Subject)
    Returns: float (the average mark)
  
  ✓ getAverageAndGrade(SubjectMarks, Subject)  ⭐ MAIN
    Returns: ['average' => 38.33, 'grade' => 'F']
  
  ✓ calculateTotalFromAverages($marks, $configs)
    Returns: float (sum of all averages)

Usage:
  $result = $averageMarksService->getAverageAndGrade($mark, $subject);
  $average = $result['average'];    // 38.33
  $grade = $result['grade'];        // 'F'

SERVICE 2: GradeLookupService
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

File: app/Services/Results/GradeLookupService.php

Main Lookup Methods:
  ✓ getGradeFromMarks(85)
    Returns: 'A'
  
  ✓ getCompetenceLevel('A')
    Returns: 'Excellent'
  
  ✓ getGradePoints('A')
    Returns: 1
  
  ✓ getDivisionFromPoints(8)  ⭐ FOR DIVISION
    Returns: ['division' => 1, 'name' => 'I', 'competence' => 'Excellent']
  
  ✓ getGPA(10, 5)  ⭐ FOR GPA
    Returns: 2.0
  
  ✓ formatDivision(1)  ⭐ FOR DISPLAY
    Returns: 'I'
  
  ✓ getGradeColor('A')
    Returns: '#00AA7A'
  
  ✓ getCompleteGradeInfo(...)
    Returns: Complete array with all info

Usage Examples:
  // Get division
  $div = $gradeLookupService->getDivisionFromPoints(8);
  echo "Division: {$div['name']}";  // "Division: I"
  
  // Get GPA
  $gpa = $gradeLookupService->getGPA(10, 5);
  echo "GPA: {$gpa}";  // "GPA: 2.0"
  
  // Get competence level
  $competence = $gradeLookupService->getCompetenceLevel('A');
  echo "Competence: {$competence}";  // "Competence: Excellent"
  
  // Format division for display
  $display = $gradeLookupService->formatDivision(3);
  echo "Division: {$display}";  // "Division: III"
