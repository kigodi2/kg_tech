HOW TO IMPORT:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1. Create schools.csv with your data
  2. Run: php artisan tinker
  3. Paste import code:

php tinker << 'EOF'
$file = fopen('schools.csv', 'r');
$header = fgetcsv($file);

while (($row = fgetcsv($file)) !== false) {
    if (empty(array_filter($row))) continue;
    
    $region = App\Models\Region::where('code', $row[3])->first();
    $district = App\Models\District::where('code', $row[4])->first();
    
    if (!$region || !$district) continue;
    
    App\Models\School::updateOrCreate(
        ['registration_number' => $row[0]],
        [
            'code' => $row[0],
            'registration_number' => $row[0],
            'name' => $row[1],
            'ownership' => strtoupper($row[2]),
            'region_id' => $region->id,
            'district_id' => $district->id,
            'school_type' => 'SECONDARY',
            'is_active' => true
        ]
    );
}

echo "Done!\n";
EOF
